    int main(){ __LAPACK_double_complex v{};cblas_zgemm(CblasColMajor,CblasNoTrans,CblasNoTrans,1,1,1,&v,&v,1,&v,1,&v,&v,1); }"

  set_source_files_properties(src/apple_modern_blas.cpp PROPERTIES COMPILE_DEFINITIONS ACCELERATE_NEW_LAPACK)

if(SDFT_ENABLE_OPENMP)
  find_package(OpenMP QUIET)
  set(_SDFT_OMP_CANDIDATE FALSE)
  set(_SDFT_OMP_REQUIRED_FLAGS "")
  set(_SDFT_OMP_REQUIRED_FLAGS_STRING "")
  set(_SDFT_OMP_REQUIRED_LIBRARIES "")
  set(_SDFT_OMP_REQUIRED_INCLUDES "")
  if(OpenMP_CXX_FOUND)
    set(_SDFT_OMP_CANDIDATE TRUE)
    set(_SDFT_OMP_REQUIRED_FLAGS "${OpenMP_CXX_FLAGS}")
    set(_SDFT_OMP_REQUIRED_FLAGS_STRING "${OpenMP_CXX_FLAGS}")
    if(TARGET OpenMP::OpenMP_CXX)
      set(_SDFT_OMP_REQUIRED_LIBRARIES OpenMP::OpenMP_CXX)
    else()
      set(_SDFT_OMP_REQUIRED_LIBRARIES "${OpenMP_CXX_LIBRARIES}")
    endif()
  elseif(APPLE)
    # AppleClang accepts libomp pragmas only when the preprocessor is given
    # -fopenmp. Merely adding omp.h and libomp to the target silently leaves
    # every #pragma omp inactive.
    find_path(SDFT_LIBOMP_INCLUDE_DIR omp.h HINTS /opt/homebrew/opt/libomp/include /usr/local/opt/libomp/include)
    find_library(SDFT_LIBOMP_LIBRARY NAMES omp libomp HINTS /opt/homebrew/opt/libomp/lib /usr/local/opt/libomp/lib)
    if(SDFT_LIBOMP_INCLUDE_DIR AND SDFT_LIBOMP_LIBRARY)
      set(_SDFT_OMP_CANDIDATE TRUE)
      set(_SDFT_OMP_REQUIRED_INCLUDES "${SDFT_LIBOMP_INCLUDE_DIR}")
      set(_SDFT_OMP_REQUIRED_LIBRARIES "${SDFT_LIBOMP_LIBRARY}")
      if(CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang")
        # Keep these as two list entries for target_compile_options().
        set(_SDFT_OMP_REQUIRED_FLAGS -Xpreprocessor -fopenmp)
        set(_SDFT_OMP_REQUIRED_FLAGS_STRING "-Xpreprocessor -fopenmp")
      else()
        set(_SDFT_OMP_REQUIRED_FLAGS -fopenmp)
        set(_SDFT_OMP_REQUIRED_FLAGS_STRING "-fopenmp")
      endif()
    endif()
  endif()

  # Do not report OpenMP based on header/library discovery alone. This probe
  # compiles, links and executes a two-thread parallel region during configure.
  if(_SDFT_OMP_CANDIDATE)
    include(CheckCXXSourceRuns)
    set(_SDFT_OLD_REQUIRED_FLAGS "${CMAKE_REQUIRED_FLAGS}")
    set(_SDFT_OLD_REQUIRED_LIBRARIES "${CMAKE_REQUIRED_LIBRARIES}")
    set(_SDFT_OLD_REQUIRED_INCLUDES "${CMAKE_REQUIRED_INCLUDES}")
    set(_SDFT_OLD_REQUIRED_QUIET "${CMAKE_REQUIRED_QUIET}")
    set(CMAKE_REQUIRED_FLAGS "${_SDFT_OMP_REQUIRED_FLAGS_STRING}")
    set(CMAKE_REQUIRED_LIBRARIES "${_SDFT_OMP_REQUIRED_LIBRARIES}")
    set(CMAKE_REQUIRED_INCLUDES "${_SDFT_OMP_REQUIRED_INCLUDES}")
    set(CMAKE_REQUIRED_QUIET TRUE)
    unset(SDFT_OPENMP_RUNTIME_WORKS CACHE)
    check_cxx_source_runs([=[
      #include <omp.h>
      int main() {
        omp_set_dynamic(0);
        omp_set_num_threads(2);
        int team = 0;
        #pragma omp parallel reduction(+:team)
        { team += 1; }
        return team == 2 ? 0 : 1;
      }
    ]=] SDFT_OPENMP_RUNTIME_WORKS)
    set(CMAKE_REQUIRED_FLAGS "${_SDFT_OLD_REQUIRED_FLAGS}")
    set(CMAKE_REQUIRED_LIBRARIES "${_SDFT_OLD_REQUIRED_LIBRARIES}")
    set(CMAKE_REQUIRED_INCLUDES "${_SDFT_OLD_REQUIRED_INCLUDES}")
    set(CMAKE_REQUIRED_QUIET "${_SDFT_OLD_REQUIRED_QUIET}")
  endif()

  if(_SDFT_OMP_CANDIDATE AND SDFT_OPENMP_RUNTIME_WORKS)
    if(TARGET OpenMP::OpenMP_CXX AND OpenMP_CXX_FOUND)
      target_link_libraries(sdft_core PUBLIC OpenMP::OpenMP_CXX)
    else()
      target_compile_options(sdft_core PUBLIC ${_SDFT_OMP_REQUIRED_FLAGS})
      target_include_directories(sdft_core PUBLIC ${_SDFT_OMP_REQUIRED_INCLUDES})
      target_link_libraries(sdft_core PUBLIC ${_SDFT_OMP_REQUIRED_LIBRARIES})
    endif()
    target_compile_definitions(sdft_core PUBLIC SDFT_HAS_OPENMP=1)
    set(SDFT_HAS_OPENMP TRUE CACHE BOOL "OpenMP available for Sparse-DFT" FORCE)
    message(STATUS "Sparse-DFT OpenMP enabled (compile/link/runtime two-thread probe passed)")
  else()
    set(SDFT_HAS_OPENMP FALSE CACHE BOOL "OpenMP available for Sparse-DFT" FORCE)
    if(_SDFT_OMP_CANDIDATE)
      message(STATUS "Sparse-DFT OpenMP candidate failed compile/link/runtime probe; using serial kernels")
    else()
      message(STATUS "Sparse-DFT OpenMP compiler/runtime not found; using serial kernels")
    endif()
  endif()
else()
  set(SDFT_HAS_OPENMP FALSE CACHE BOOL "OpenMP available for Sparse-DFT" FORCE)
endif()
